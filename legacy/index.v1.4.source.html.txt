function canMoveAs(piece,moveType,tr,tc,pieces,board){
  const rows=board.rows,cols=board.cols;
  const at=(r,c)=>pieces.find(p=>p.row===r&&p.col===c);
  const outside=(r,c)=>board.outside.some(b=>b[0]===r&&b[1]===c);
  const wall=(r,c)=>board.walls.some(b=>b[0]===r&&b[1]===c);
  const blockedAt=(r,c)=>outside(r,c)||wall(r,c);
  const inBounds=(r,c)=>r>=0&&r<rows&&c>=0&&c<cols;
  if(!inBounds(tr,tc))return false;
  if(blockedAt(tr,tc))return false;
  const pt=at(tr,tc);
  if(pt&&pt.color===piece.color)return false;
  const dr=tr-piece.row,dc=tc-piece.col;
  const adr=Math.abs(dr),adc=Math.abs(dc);
  const pathClear=(dr,dc)=>{
    let r=piece.row+Math.sign(dr),c=piece.col+Math.sign(dc);
    while(r!==tr||c!==tc){
      if(!inBounds(r,c))return false;
      if(blockedAt(r,c))return false;
      if(at(r,c))return false;
      r+=Math.sign(dr);c+=Math.sign(dc);
    }
    return true;
  };
  switch(moveType){
    case 'pawn':{
      const dir=piece.color==='light'?-1:1;
      const sr=(typeof piece.startRow==='number')?piece.startRow:(piece.color==='light'?rows-2:1);
      if(dc===0&&dr===dir)return !pt;
      if(dc===0&&dr===2*dir&&piece.row===sr){
        const mid=piece.row+dir;
        if(!inBounds(mid,piece.col))return false;
        if(at(mid,piece.col))return false;
        if(blockedAt(mid,piece.col))return false;
        return !pt;
      }
      if(adc===1&&dr===dir)return !!pt;
      return false;
    }
    case 'pawnBack':{const dir=piece.color==='light'?-1:1;if(dc===0&&dr===-dir)return !pt;return false;}
    case 'guard':{const dir=piece.color==='light'?-1:1;if(adr>1||adc>1)return false;if(adr===0&&adc===0)return false;if(dr===-dir&&dc===0)return false;return true;}
    case 'advisor':return adr===1&&adc===1;
    case 'elephant':return adr===2&&adc===2;
    case 'camel':return (adr===3&&adc===1)||(adr===1&&adc===3);
    case 'cannon':{
      if(dr!==0&&dc!==0)return false;
      if(dr===0&&dc===0)return false;
      const maxRange=4+(piece.cannonRangeBonus||0);
      if(adr+adc>maxRange)return false;
      let b=0;
      let r=piece.row+Math.sign(dr),c=piece.col+Math.sign(dc);
      while(r!==tr||c!==tc){
        if(!inBounds(r,c))return false;
        if(blockedAt(r,c))return false;
        if(at(r,c))b++;
        r+=Math.sign(dr);c+=Math.sign(dc);
      }
      if(b===0)return !pt;
      if(b===1)return !!pt;
      return false;
    }
    case 'knight':return (adr===2&&adc===1)||(adr===1&&adc===2);
    case 'bishop':if(adr!==adc||adr===0)return false;return pathClear(dr,dc);
    case 'rook':if(dr===0&&dc===0)return false;if(dr!==0&&dc!==0)return false;return pathClear(dr,dc);
    case 'rook2':if(dr!==0&&dc!==0)return false;if(dr===0&&dc===0)return false;if(adr+adc>2)return false;return pathClear(dr,dc);
    case 'queen':if(dr===0&&dc===0)return false;if(dr===0||dc===0||adr===adc)return pathClear(dr,dc);return false;
    case 'king':return adr<=1&&adc<=1&&(adr+adc>0);
  }
  return false;
}
function pieceCanMove(piece,tr,tc,pieces,board){
  if(canMoveAs(piece,piece.type,tr,tc,pieces,board))return true;
  if(piece.extraMoveTypes){
    for(const et of piece.extraMoveTypes){
      if(canMoveAs(piece,et,tr,tc,pieces,board))return true;
    }
  }
  return false;
}
function getLegalMoves(piece,pieces,board){
  const moves=[];
  for(let r=0;r<board.rows;r++){
    for(let c=0;c<board.cols;c++){
      if(r===piece.row&&c===piece.col)continue;
      if(pieceCanMove(piece,r,c,pieces,board))moves.push({row:r,col:c});
    }
  }
  return moves;
}
function hasAnyLegalMove(pieces,color,board){
  const mine=pieces.filter(p=>p.color===color);
  for(const p of mine)if(getLegalMoves(p,pieces,board).length>0)return true;
  return false;
}

function generateBoard(seed,floor,totalFloors){
  const rng=mulberry32(seed);
  const t=totalFloors>1?floor/(totalFloors-1):0;
  const base=8+Math.round(t*4);
  const rows=Math.max(9,Math.min(12,base+(rng()<0.5?1:0)));
  const cols=Math.max(9,Math.min(12,base+(rng()<0.5?1:0)));
  const safeRows= rows<=9 ? 2 : rows<=10 ? 3 : 4;
  const shape=pickFrom(rng,['diamond','hourglass','cross','fortress','random','stripes','scatter','lake','canyon','islands']);
  const outsideSet=new Set();
  const topSafe=safeRows,bottomSafe=rows-safeRows;
  function addSym(r,c){
    const positions=[[r,c],[rows-1-r,c],[r,cols-1-c],[rows-1-r,cols-1-c]];
    for(const [rr,cc] of positions){
      if(rr<0||rr>=rows||cc<0||cc>=cols)continue;
      if(rr<topSafe||rr>=bottomSafe)continue;
      outsideSet.add(rr+','+cc);
    }
  }
  const cR=(rows-1)/2,cC=(cols-1)/2,maxDist=cR+cC;
  const halfC=Math.ceil(cols/2);
  for(let r=topSafe;r<bottomSafe;r++){
    for(let c=0;c<halfC;c++){
      const dr=Math.abs(r-cR),dc=Math.abs(c-cC),dist=dr+dc;
      const edgeR=Math.min(r,rows-1-r);
      const edgeC=Math.min(c,cols-1-c);
      let isOut=false;
      switch(shape){
        case 'diamond':if(dist>maxDist*0.75)isOut=rng()<0.85;break;
        case 'hourglass':if(Math.abs(dr-dc)>Math.min(rows,cols)*0.34)isOut=true;break;
        case 'cross':if(dr>rows*0.35&&dc>cols*0.35)isOut=true;break;
        case 'fortress':if(dr>rows*0.4&&dc>cols*0.4)isOut=true;break;
        case 'random':if(dist>maxDist*0.5&&rng()<0.6)isOut=true;break;
        case 'stripes':if(dc>cols*0.3&&(r+c)%3===0&&rng()<0.7)isOut=true;break;
        case 'scatter':if(dist>maxDist*0.4&&rng()<0.5)isOut=true;break;
        case 'lake':if(dist<maxDist*0.35&&rng()<0.8)isOut=true;else if(dist>maxDist*0.65&&rng()<0.5)isOut=true;break;
        case 'canyon':if(Math.abs(c-cC)<1.6&&edgeR>1&&rng()<0.7)isOut=true;break;
        case 'islands':if(edgeR>1&&edgeC>1&&rng()<0.22)isOut=true;if(dist>maxDist*0.7&&rng()<0.6)isOut=true;break;
      }
      if(isOut)addSym(r,c);
    }
  }
  const maxHoles=Math.floor(rows*cols*0.35);
  if(outsideSet.size>maxHoles){
    const arr=Array.from(outsideSet);
    for(let i=arr.length-1;i>0;i--){const j=Math.floor(rng()*(i+1));[arr[i],arr[j]]=[arr[j],arr[i]];}
    outsideSet.clear();
    for(let i=0;i<maxHoles;i++)outsideSet.add(arr[i]);
  }
  const outside=Array.from(outsideSet).map(k=>k.split(',').map(Number));
  return {outside,walls:[],shape,rows,cols,protectRows:safeRows};
}

function generateMap(seed){
  const rng=mulberry32(seed);
  const floors=5+Math.floor(rng()*11);
  const nodes=[],connections=[],layouts=[];
  for(let f=0;f<floors;f++){
    if(f===0||f===floors-1)layouts.push(1);
    else layouts.push(1+Math.floor(rng()*3));
  }
  const modPool=['none','promote','promote','chaos','chaos','chaos','rush','rush','fog','fog','rest','bag'];
  for(let f=0;f<floors;f++){
    const nc=layouts[f];
    const isBoss=f===floors-1,isFirst=f===0;
    for(let c=0;c<nc;c++){
      const nodeSeed=Math.floor(rng()*1000000);
      let modId;
      if(isFirst)modId='none';
      else if(isBoss)modId='boss';
      else modId=pickFrom(rng,modPool);
      const diff=isBoss?5:Math.min(5,1+Math.floor((f/Math.max(1,floors-1))*4));
      nodes.push({id:`f${f}c${c}`,floor:f,column:c,totalCols:nc,modifier:modId,seed:nodeSeed,diff,isBoss,isFirst});
    }
  }
  for(let f=0;f<floors-1;f++){
    const cc=layouts[f],nc=layouts[f+1];
    for(let c=0;c<cc;c++){
      const cur=nodes.find(n=>n.floor===f&&n.column===c);
      const ratio=cc>1?c/(cc-1):0.5;
      const ti=Math.round(ratio*(nc-1));
      const targets=new Set([ti]);
      if(nc>1){
        if(rng()<0.5)targets.add(Math.max(0,ti-1));
        if(rng()<0.5)targets.add(Math.min(nc-1,ti+1));
      }
      for(const t of targets){
        const to=nodes.find(n=>n.floor===f+1&&n.column===t);
        if(to)connections.push({from:cur.id,to:to.id});
      }
    }
    for(let c=0;c<nc;c++){
      const to=nodes.find(n=>n.floor===f+1&&n.column===c);
      const hasIn=connections.some(cn=>cn.to===to.id);
      if(!hasIn){
        const ratio=nc>1?c/(nc-1):0.5;
        const si=Math.round(ratio*(cc-1));
        const from=nodes.find(n=>n.floor===f&&n.column===si);
        if(from)connections.push({from:from.id,to:to.id});
      }
    }
  }
  return {nodes,floors,connections,layouts};
}

    if(numLayers>=4&&back-6>=2)layers.push(back-6);
    return layers;
  }
  function placeSide(color,backRow,direction,numLayers){
    const backCols=[];
    for(let c=0;c<cols;c++)if(isValid(backRow,c))backCols.push(c);
    if(backCols.length<2)return;
    const layerCounts=computeSideLayers(numLayers,backCols.length);
    const backList=backRankList(Math.min(layerCounts[0],backCols.length));
    const backSel=centeredColumns(backCols,backList.length);
    for(let i=0;i<backSel.length;i++){
      pieces.push({type:backList[i],color,row:backRow,col:backSel[i]});
    }
    for(let l=1;l<layerCounts.length;l++){
      const r=backRow+direction*l;
      if(r<0||r>=rows)continue;
      const validCols=[];
      for(let c=0;c<cols;c++)if(isValid(r,c))validCols.push(c);
      if(!validCols.length)continue;
      const count=Math.min(layerCounts[l],validCols.length);
      const sel=centeredColumns(validCols,count);
      const types=(l===1)?newPiecesLayer(count):pawnLayer(count);
      for(let i=0;i<sel.length;i++){
        const type=types[i]||'pawn';
        const p={type,color,row:r,col:sel[i]};
        if(type==='pawn')p.startRow=r;
        pieces.push(p);
      }
    }
  }
  placeSide('dark',0,1,darkLayers);
  placeSide('light',rows-1,-1,lightLayers);
  if(playerCount===3){
    const cr=Math.floor(rows/2),cc=Math.floor(cols/2);
    const wilds=[
      {r:cr-1,c:cc-1,t:'guard'},{r:cr-1,c:cc,t:'pawn'},{r:cr-1,c:cc+1,t:'guard'},
      {r:cr,c:cc-1,t:'camel'},{r:cr,c:cc,t:'king'},{r:cr,c:cc+1,t:'queen'}
    ];
    for(const w of wilds){
      if(isValid(w.r,w.c)){
        const p={type:w.t,color:'wild',row:w.r,col:w.c};
        if(w.t==='pawn')p.startRow=w.r;
        pieces.push(p);
      }
    }
  }
  if(isBoss){
    const enemies=playerCount===3?['dark','wild']:['dark'];
    const extraOptions=['knight','bishop','rook','queen','king','camel','elephant'];
    for(const p of pieces){
      if(!enemies.includes(p.color))continue;
      const isBack=p.row<board.protectRows;
      const isGuard=p.type==='guard'||p.type==='advisor';
      if(isBack||isGuard){
        const numExtra=rng()<0.35?2:1;
        const types=new Set();
        let safety=0;
        while(types.size<numExtra&&safety<12){
          safety++;
          const tt=pickFrom(rng,extraOptions);
          if(tt!==p.type)types.add(tt);
        }
        p.extraMoveTypes=Array.from(types);
      }
    }
  }
  if(mutations&&mutations.length)applyMutationsToPieces(pieces,mutations,playerColor||'light');
  /* One-shot bonuses */
  if(oneShot){
    if(oneShot.promoteOne){
      const p=pieces.find(p=>p.type==='pawn'&&p.color==='light');
      if(p)p.type='queen';
    }
    if(oneShot.extraRook){
      const row=rows-1;
      for(let c=0;c<cols;c++){
        if(pieces.some(p=>p.row===row&&p.col===c))continue;
        if(!isValid(row,c))continue;
        pieces.push({type:'rook',color:'light',row,col:c});
        break;
      }
    }
    if(oneShot.extraPawn){
      const row=rows-2;
      for(let c=0;c<cols;c++){
        if(pieces.some(p=>p.row===row&&p.col===c))continue;
        if(!isValid(row,c))continue;
        pieces.push({type:'pawn',color:'light',row,col:c,startRow:row});
        break;
      }
    }
  }
  return pieces;
}
function computeLayout(cw,ch,rows,cols){
  const m=Math.min(cw,ch);
  const margin=Math.max(8,Math.min(m*0.04,30));
  const availW=cw-margin*2,availH=ch-margin*2;
  const square=Math.max(12,Math.floor(Math.min(availW/cols,availH/rows)));
  const boardW=square*cols,boardH=square*rows;
  const originX=Math.floor((cw-boardW)/2),originY=Math.floor((ch-boardH)/2);
  return {square,boardW,boardH,originX,originY,rows,cols};
}

function screenToSquare(px,py,layout){
  const col=Math.floor((px-layout.originX)/layout.square);
  const row=Math.floor((py-layout.originY)/layout.square);
  if(row<0||row>=layout.rows||col<0||col>=layout.cols)return null;
  return {row,col};
}

function botChooseMove(pieces,color,board){
  const mine=pieces.filter(p=>p.color===color);
  const enemies=pieces.filter(p=>p.color!==color);
  if(!mine.length||!enemies.length)return null;
  const candidates=[];
  for(const p of mine){
    if(p.type==='king'&&mine.length<=2)continue;
    const moves=getLegalMoves(p,pieces,board);
    for(const m of moves){
      const occ=pieces.find(o=>o.row===m.row&&o.col===m.col);
      let score=0;
      if(occ){const v={pawn:10,guard:18,advisor:18,elephant:22,camel:30,cannon:40,knight:30,bishop:30,rook:50,queen:90,king:1000};score+=v[occ.type]||20;}
      const dc=Math.abs(m.row-(board.rows-1)/2)+Math.abs(m.col-(board.cols-1)/2);
      score+=(Math.max(board.rows,board.cols)-dc);
      const ek=enemies.find(e=>e.type==='king');
      if(ek){const d=Math.abs(m.row-ek.row)+Math.abs(m.col-ek.col);score+=Math.max(0,10-d);}
      score+=Math.random()*4;
      candidates.push({piece:p,target:m,score});
    }
  }
  if(!candidates.length)return null;
  candidates.sort((a,b)=>b.score-a.score);
  return candidates[0];
}

function mutationExtrasFor(type,mutCount){
  const extras=[];
  if(type==='knight'&&mutCount.knightBishop)extras.push('bishop');
  if(type==='rook'&&mutCount.towerDiag)extras.push('advisor');
  if(type==='king'&&mutCount.kingRook)extras.push('rook2');

    if(extras.length){
      p.extraMoveTypes=(p.extraMoveTypes||[]).concat(extras);
      p.mutationExtras=extras.slice();
    }
  }
}

function playMoveTrail(engine,sprite,fromX,fromY,toX,toY,color){
  const steps=10;
  for(let i=0;i<steps;i++){
    setTimeout(()=>{
      const t=i/steps;
      const x=fromX+(toX-fromX)*t;
      const y=fromY+(toY-fromY)*t;
      engine.burst(x,y,{count:3,color,speed:[20,60],life:[0.35,0.7],size:[2,4]});
    }, i*28);
  }
}
function playChaosCollapse(engine,layout,board,newWalls,addLog){
  if(!engine)return;
  const accent='#ff6d43',dark='#e83d33';
  const affected=[];
  for(const [r,c] of newWalls){
    affected.push([r,c]);
    for(const [dr,dc] of [[1,0],[-1,0],[0,1],[0,-1]]){
      const rr=r+dr,cc=c+dc;
      if(rr<0||rr>=board.rows||cc<0||cc>=board.cols)continue;
      if(board.outside.some(b=>b[0]===rr&&b[1]===cc))continue;
      if(board.walls.some(b=>b[0]===rr&&b[1]===cc))continue;
      affected.push([rr,cc]);
    }
  }
  engine.shake(18,0.55);
  newWalls.forEach(([r,c])=>{
    if(!layout)return;
    const {x,y}=squareToScreen(r,c,layout);
    engine.burst(x,y,{count:40,color:'#fff',speed:[120,420],size:[2,6],life:[0.3,0.7]});
    engine.burst(x,y,{count:30,color:accent,speed:[80,300],size:[2,5],life:[0.4,0.9]});
    engine.burst(x,y,{count:20,color:dark,speed:[60,200],size:[2,4],life:[0.5,1.1],gravity:80});
  });
  affected.forEach(([r,c],i)=>{
    setTimeout(()=>{
      if(!layout)return;
      const {x,y}=squareToScreen(r,c,layout);
      engine.burst(x,y,{count:8,color:accent,speed:[60,180],size:[1,4],life:[0.3,0.7],gravity:60});
    }, 80+i*40);
  });
  if(addLog)addLog('¡Se derrumba el borde!');
}
function playCoronation(engine,layout,sprite,onDone){
  if(!sprite||!engine){if(onDone)onDone();return;}
  const cx=sprite.x,cy=sprite.y;
  let count=0;
  const maxFrames=8;
  function step(){
    if(count>=maxFrames){
      engine.shake(10,0.35);
      engine.burst(cx,cy,{count:50,color:'#ff6d43',speed:[80,300],size:[2,5],life:[0.4,0.9]});
      engine.burst(cx,cy,{count:30,color:'#f0d9a8',speed:[60,240],size:[2,5],life:[0.5,1.0]});
      sprite.x=cx;sprite.y=cy;
      if(onDone)onDone();
      return;
    }
    sprite.x=cx+(Math.random()*10-5);
    sprite.y=cy+(Math.random()*10-5);
    engine.shake(4,0.08);
    engine.burst(cx-6-Math.random()*6,cy+(Math.random()*8-4),{count:3,color:'#e83d33',speed:[20,80],life:[0.2,0.5],size:[2,4]});
    engine.burst(cx+6+Math.random()*6,cy+(Math.random()*8-4),{count:3,color:'#8fd46a',speed:[20,80],life:[0.2,0.5],size:[2,4]});
    count++;
    setTimeout(step,50);
  }
  step();
}

}
function playSpawnWave(engine,pieces,layout){
  if(!engine||!layout)return;
  /* Una onda de partículas que barre todas las piezas al inicio */
  const sorted=pieces.slice().sort((a,b)=>a.row-b.row);
  sorted.forEach((p,i)=>{
    if(!p.sprite)return;
    const s=p.sprite;
    setTimeout(()=>{
      engine.burst(s.x,s.y,{count:6,color:p.color==='light'?'#8fd46a':'#ff6d43',speed:[20,70],life:[0.3,0.6],size:[2,4]});
    }, i*18);
  });
  engine.shake(6,0.4);
}
function playVictoryAnim(engine,pieces,humanColor,onDone){
  if(!engine){if(onDone)onDone();return;}
  const enemies=pieces.filter(p=>p.color!==humanColor&&p.sprite);
  enemies.forEach((p,i)=>{
    const s=p.sprite;
    setTimeout(()=>{
      gsap.to(s,{y:s.y-180,scale:0.4,alpha:0,duration:1.0,ease:'power2.in'});
      engine.burst(s.x,s.y,{count:30,color:'#8fd46a',speed:[-60,-260],size:[2,5],life:[0.6,1.3],gravity:-40});
    }, i*60);
  });
  engine.shake(8,0.5);
  setTimeout(()=>{if(onDone)onDone();},Math.min(enemies.length*60+1100,1600));
}

}

function MapScreen({onBack,onSelectNode,mapState,onNewRun}){
  const {nodes,floors,connections}=mapState.map;
  const currentFloor=mapState.currentFloor;
  const completed=mapState.completed;
  const rowH=100,padTop=40,padBottom=60;
  const canvasH=padTop+(floors-1)*rowH+padBottom;
  const colX=(col,totalCols)=>totalCols===1?50:20+(col/(totalCols-1))*60;
  const floorY=(f)=>(floors-1-f)*rowH+padTop;
  const posOf=(node)=>({x:colX(node.column,node.totalCols),y:floorY(node.floor)});
  function isAvailable(node){
    if(completed[node.id])return false;
    if(node.floor!==currentFloor)return false;
    if(node.floor===0)return true;
    const preds=connections.filter(cn=>cn.to===node.id);
    return preds.some(cn=>completed[cn.from]);
  }
  return React.createElement('div',{className:'screen active'},
    React.createElement('div',{className:'topbar'},
      React.createElement('button',{className:'back-btn',onClick:onBack},
        React.createElement(Icon,{name:'menu',size:14}),
        React.createElement('span',null,'Menu')),
      React.createElement('div',null,
        React.createElement('div',{className:'title'},'Elige tu ruta'),
        React.createElement('div',{className:'sub'},'Piso '+(currentFloor+1)+' de '+floors)),
      React.createElement('div',{className:'spacer'}),
      React.createElement('button',{className:'icon-btn',onClick:onNewRun},
        React.createElement(Icon,{name:'refresh',size:14}),
        React.createElement('span',null,'Nueva'))
    ),
    React.createElement('div',{className:'map-wrap'},
      React.createElement('div',{className:'map-canvas',style:{height:canvasH+'px'}},
        React.createElement('svg',{className:'map-connectors',viewBox:`0 0 100 ${canvasH}`,preserveAspectRatio:'none'},
          connections.map((cn,i)=>{
            const from=nodes.find(n=>n.id===cn.from);
            const to=nodes.find(n=>n.id===cn.to);
            if(!from||!to)return null;
            const fp=posOf(from),tp=posOf(to);
            const done=completed[from.id]&&completed[to.id];
            return React.createElement('line',{key:'cn'+i,className:'line'+(done?' done':''),x1:fp.x,y1:fp.y,x2:tp.x,y2:tp.y,vectorEffect:'non-scaling-stroke'});
          })
        ),
        Array.from({length:floors}).map((_,f)=>
          React.createElement('div',{key:'fl-'+f,className:'map-floor-tag',style:{top:floorY(f)+'px'}},'PISO '+(f+1))
        ),
        nodes.map(node=>{
          const mod=MODIFIERS[node.modifier];
          const available=isAvailable(node);
          const done=!!completed[node.id];
          const locked=!available&&!done;
          const pos=posOf(node);
          let extraCls='';
          if(node.modifier==='rest')extraCls=' rest';
          else if(node.modifier==='bag')extraCls=' bag';
          return React.createElement('div',{
            key:node.id,
            className:'map-node'+(locked?' locked':'')+(available?' available':'')+(done?' completed':'')+(node.isBoss?' boss':'')+extraCls,
            style:{left:pos.x+'%',top:pos.y+'px'},
            onClick:()=>available&&onSelectNode(node)
          },
            React.createElement('div',{className:'node-inner'},
              done?React.createElement(Icon,{name:'check',size:node.isBoss?20:16}):
              React.createElement(Icon,{name:node.isBoss?'boss':mod.icon,size:node.isBoss?20:16})
            )
          );
        })
      ),
      React.createElement('div',{className:'map-legend'},
        React.createElement('span',null,React.createElement('span',{className:'lg-dot lg-avail'}),'Disponible'),
        React.createElement('span',null,React.createElement('span',{className:'lg-dot lg-done'}),'Completado'),
        React.createElement('span',null,React.createElement('span',{className:'lg-dot lg-lock'}),'Bloqueado')
      )
    )
  );
}

function GameScreen({node,match,mapState,settings,setSettings,onExit,onAbandon,onWin,onRestartNode,mutations,oneShotBonuses}){
  const canvasRef=useRef(null);
  const engineRef=useRef(null);
  const piecesRef=useRef([]);
  const layoutRef=useRef(null);
  const dragRef=useRef(null);
  const selectedRef=useRef(null);
  const inspectRef=useRef(null);
  const legalRef=useRef(new Set());
  const legalColorRef=useRef('legal');
  const boardRef=useRef(null);
  const turnColorRef=useRef('light');
  const currentIdxRef=useRef(0);
  const matchRef=useRef(match);
  const threatRef=useRef(new Set());
  const effectRef=useRef({pending:[]});
  const chaosCounterRef=useRef(0);
  const logRef=useRef(null);
  const [movesThisTurn,setMovesThisTurn]=useState(0);
  const [currentIdx,setCurrentIdx]=useState(0);
  const [loaded,setLoaded]=useState(false);
  const [paused,setPaused]=useState(false);
  const [showHTP,setShowHTP]=useState(false);
  const [log,setLog]=useState([]);
  const [botThinking,setBotThinking]=useState(false);
  const [eliminated,setEliminated]=useState({});
  const [endState,setEndState]=useState(null);
  const [inspectInfo,setInspectInfo]=useState(null);
  const [threatCount,setThreatCount]=useState(0);
  const [chaosTick,setChaosTick]=useState(0);
  const [layerInfo,setLayerInfo]=useState({dark:0,light:0});
  const [turnFlashKey,setTurnFlashKey]=useState(0);
  const [turnFlashVisible,setTurnFlashVisible]=useState(false);
  const mod=MODIFIERS[node.modifier];
  const players=match.players;
  /* Moves per player: mutations apply only to 'light' color, others get base */
  const mutationBearingColor='light';
  const mutCount=countMutations(mutations||[]);
  const furyStack=Math.min(3,mutCount.fury||0);
  const oneShotFury=oneShotBonuses&&oneShotBonuses.furyNow?1:0;
  const movesForColor=(color)=>{
    const base=mod.id==='rush'?2:1;
    if(color===mutationBearingColor)return base+furyStack+oneShotFury;
    return base;
  };
  const currentPlayer=players[currentIdx];
  const movesPerTurn=currentPlayer?movesForColor(currentPlayer.color):1;
  const isBossNode=node.isBoss;
  const totalFloors=mapState.map.floors;
  const addLog=useCallback((m)=>setLog(p=>[...p.slice(-40),m]),[]);
  useEffect(()=>{currentIdxRef.current=currentIdx;},[currentIdx]);
  useEffect(()=>{matchRef.current=match;},[match]);
  useEffect(()=>{
    const p=players[currentIdx];
    turnColorRef.current=p?p.color:'light';
    if(p&&p.kind==='human'&&loaded&&!endState){
      setTurnFlashKey(k=>k+1);
      setTurnFlashVisible(true);
      const t=setTimeout(()=>setTurnFlashVisible(false),1000);
      return ()=>clearTimeout(t);
    }
  },[currentIdx,players,loaded,endState]);
  useEffect(()=>{const el=logRef.current;if(el)el.scrollTop=el.scrollHeight;},[log]);
  useEffect(()=>{
    if(!loaded)return;
    const cur=players[currentIdx];
    if(!cur)return;
    const threats=getThreatenedOwnPieces(piecesRef.current,cur.color,boardRef.current);
    threatRef.current=threats;
    setThreatCount(threats.size);
  },[currentIdx,loaded,players]);
  useEffect(()=>{
    const engine=new Engine.Engine(canvasRef.current);
    engineRef.current=engine;
    const board=generateBoard(node.seed,node.floor,totalFloors);
    boardRef.current=board;
    layoutRef.current=computeLayout(engine.W||400,engine.H||400,board.rows,board.cols);
    const boardLayer=makeBoardLayer(layoutRef,boardRef,legalRef,legalColorRef,piecesRef,turnColorRef,node.modifier,threatRef,effectRef);
    engine.sprites.push(boardLayer);
    engine.onResize=()=>{
      layoutRef.current=computeLayout(engine.W,engine.H,board.rows,board.cols);
      for(const p of piecesRef.current){
        if(!p.sprite)continue;
        const {x,y}=squareToScreen(p.row,p.col,layoutRef.current);
        if(!Number.isFinite(x)||!Number.isFinite(y))continue;
        p.sprite.x=x;p.sprite.y=y;
        p.sprite.w=layoutRef.current.square*0.82;
        p.sprite.h=layoutRef.current.square*0.82;
      }
    };
    (async ()=>{
      const pieces=placePieces(board,match.playerCount,isBossNode,node.seed,mutations,'light',oneShotBonuses);
      const style=getComputedStyle(document.documentElement);
      const colorMap={light:style.getPropertyValue('--lightPiece').trim(),dark:style.getPropertyValue('--darkPiece').trim(),wild:style.getPropertyValue('--wildPiece').trim()};
      const types=['pawn','guard','advisor','elephant','camel','cannon','rook','knight','bishop','queen','king'];
      const images={};
      for(const t of types)for(const c of ['light','dark','wild'])images[c+'|'+t]=await SVG_PIECES.load(t,colorMap[c]);
      for(const p of pieces){
        const img=images[p.color+'|'+p.type];
        if(!img)continue;
        const layout=layoutRef.current;
        if(!layout||!Number.isFinite(layout.square))continue;
        if(!Number.isFinite(p.row)||!Number.isFinite(p.col))continue;
        const {x,y}=squareToScreen(p.row,p.col,layout);
        if(!Number.isFinite(x)||!Number.isFinite(y))continue;
        const sprite=new Engine.Sprite({x,y,image:img,w:layout.square*0.82,h:layout.square*0.82,alpha:0,scale:0.4,z:10});
        p.sprite=sprite;
        engine.add(sprite);
        gsap.to(sprite,{alpha:1,scale:1,duration:0.4,ease:'power2.out'});
      }
      piecesRef.current=pieces;
      const cur=players[0];
      if(cur){
        const threats=getThreatenedOwnPieces(pieces,cur.color,board);
        threatRef.current=threats;
        setThreatCount(threats.size);
      }
      const darkRows=[...new Set(pieces.filter(p=>p.color==='dark').map(p=>p.row))].sort();
      const lightRows=[...new Set(pieces.filter(p=>p.color==='light').map(p=>p.row))].sort();
      setLayerInfo({dark:darkRows.length,light:lightRows.length});
      const mutDesc=mutations&&mutations.length?' | Muts: '+mutations.length:'';
      addLog('Piso '+(node.floor+1)+'/'+totalFloors+' ('+board.rows+'x'+board.cols+'): '+mod.name);
      addLog('Formacion: '+darkRows.length+' vs '+lightRows.length+' capas'+mutDesc);
      setLoaded(true);
      setTimeout(()=>playSpawnWave(engine,pieces,layoutRef.current),400);
    })();
    return ()=>engine.destroy();
  },[node.id]);
  useEffect(()=>{
    if(!loaded)return;
    (async ()=>{
      const style=getComputedStyle(document.documentElement);
      const colorMap={light:style.getPropertyValue('--lightPiece').trim(),dark:style.getPropertyValue('--darkPiece').trim(),wild:style.getPropertyValue('--wildPiece').trim()};
      const types=['pawn','guard','advisor','elephant','camel','cannon','rook','knight','bishop','queen','king'];
      const images={};
      for(const t of types)for(const c of ['light','dark','wild'])images[c+'|'+t]=await SVG_PIECES.load(t,colorMap[c]);
      for(const p of piecesRef.current){
        if(p.sprite&&images[p.color+'|'+p.type])p.sprite.image=images[p.color+'|'+p.type];
      }
    })();
  },[settings.theme,loaded]);
  const determineWinner=useCallback((pieces,elim)=>{
    const active=players.filter(p=>!elim[p.color]);
    if(active.length===0)return {tie:true};
    if(active.length===1){
      const p=active[0];
      const humanColors=players.filter(pl=>pl.kind==='human').map(pl=>pl.color);
      return {victory:humanColors.includes(p.color),winnerName:humanColors.includes(p.color)?null:p.name};
    }
    const kings=pieces.filter(p=>p.type==='king'&&!elim[p.color]);
    if(kings.length<=1){
      if(kings.length===0)return {tie:true};
      const w=kings[0];
      const humanColors=players.filter(pl=>pl.kind==='human').map(pl=>pl.color);
      const won=humanColors.includes(w.color);
      return {victory:won,winnerName:won?null:(players.find(pl=>pl.color===w.color)||{}).name};
    }
    const counts={};
    for(const p of active)counts[p.color]=pieces.filter(x=>x.color===p.color).length;
    const maxCount=Math.max(...Object.values(counts));
    const winners=active.filter(p=>counts[p.color]===maxCount);
    if(winners.length>1)return {tie:true};
    const w=winners[0];
    const humanColors=players.filter(pl=>pl.kind==='human').map(pl=>pl.color);
    const won=humanColors.includes(w.color);
    return {victory:won,winnerName:won?null:w.name};
  },[players]);
  const finalizeEnd=useCallback((pieces,elim)=>{
    if(endingRef.current)return;
    endingRef.current=true;
    const result=determineWinner(pieces,elim);
    const humanColor=(players.find(pl=>pl.kind==='human')||{}).color||'light';
    const engine=engineRef.current;
    const setEnd=()=>setEndState(result);
    if(!engine){setEnd();return;}
    if(result.tie){setEnd();return;}
    if(result.victory){
      playVictoryAnim(engine,pieces,humanColor,setEnd);
    } else {
      playDefeatAnim(engine,pieces,humanColor,setEnd);
    }
  },[determineWinner,players]);
  const endingRef=useRef(false);
  const checkEndGame=useCallback((pieces,elim)=>{
    const kings=pieces.filter(p=>p.type==='king'&&!elim[p.color]);
    const active=players.filter(p=>!elim[p.color]);
    if(kings.length<=1){finalizeEnd(pieces,elim);return true;}
    let anyCanMove=false;
    for(const p of active)if(hasAnyLegalMove(pieces,p.color,boardRef.current)){anyCanMove=true;break;}
    if(!anyCanMove){finalizeEnd(pieces,elim);return true;}
    return false;
  },[players,finalizeEnd]);
  const applyChaosCollapse=useCallback(()=>{
    if(mod.id!=='chaos')return;
    const board=boardRef.current;
    if(!board)return;
    const candidates=[];
    for(let r=0;r<board.rows;r++){
      for(let c=0;c<board.cols;c++){
        const edgeR=Math.min(r,board.rows-1-r);
        const edgeC=Math.min(c,board.cols-1-c);
        if(edgeR>1&&edgeC>1)continue;
        if(board.outside.some(b=>b[0]===r&&b[1]===c))continue;
        if(board.walls.some(b=>b[0]===r&&b[1]===c))continue;
        if(piecesRef.current.some(p=>p.row===r&&p.col===c))continue;
        candidates.push([r,c]);
      }
    }
    if(!candidates.length)return;
    const idx=Math.floor(Math.random()*candidates.length);
    const [r,c]=candidates[idx];
    const newWalls=[[r,c]];
    const mr=board.rows-1-r,mc=board.cols-1-c;
    if(!(mr===r&&mc===c)&&!board.walls.some(b=>b[0]===mr&&b[1]===mc)&&!board.outside.some(b=>b[0]===mr&&b[1]===mc)&&!piecesRef.current.some(p=>p.row===mr&&p.col===mc)){
      newWalls.push([mr,mc]);
    } else if(candidates.length>1){
      let idx2=Math.floor(Math.random()*candidates.length);
      while(idx2===idx)idx2=Math.floor(Math.random()*candidates.length);
      newWalls.push(candidates[idx2]);
    }
    for(const [wr,wc] of newWalls) board.walls.push([wr,wc]);
    effectRef.current.pending=newWalls.map(([r,c])=>[r,c]);
    playChaosCollapse(engineRef.current,layoutRef.current,board,newWalls,addLog);
    setTimeout(()=>{effectRef.current.pending=[];},900);
  },[mod.id,addLog]);
  const advanceTurn=useCallback((elim)=>{
    setMovesThisTurn(0);
    chaosCounterRef.current++;
    setChaosTick(chaosCounterRef.current);
    if(chaosCounterRef.current%3===0)applyChaosCollapse();
    setCurrentIdx(i=>{
      const total=matchRef.current.players.length;
      let next=(i+1)%total;
      let safety=0;
      while(elim[matchRef.current.players[next].color]&&safety<total){next=(next+1)%total;safety++;}
      return next;
    });
  },[applyChaosCollapse]);
  const executeMove=useCallback((piece,targetSq,opts={})=>{
    if(!piece||!targetSq)return false;
    if(!pieceCanMove(piece,targetSq.row,targetSq.col,piecesRef.current,boardRef.current))return false;
    const layout=layoutRef.current;
    const occupant=piecesRef.current.find(o=>o!==piece&&o.row===targetSq.row&&o.col===targetSq.col);
    let capturing=false;
    let newElim=eliminated;
    let capturingKing=false;
    let capturedKingSprite=null;
    const fromX=piece.sprite.x,fromY=piece.sprite.y;
    if(occupant){
      capturing=true;
      capturingKing=occupant.type==='king';
      capturedKingSprite=occupant.sprite;
      if(capturingKing)newElim={...eliminated,[occupant.color]:true};
      gsap.to(occupant.sprite,{scale:0.15,alpha:0,rotation:Math.PI*2,duration:0.4,ease:'power2.in',onComplete:()=>{
        occupant.sprite.visible=false;
        const idx=piecesRef.current.indexOf(occupant);
        if(idx>=0)piecesRef.current.splice(idx,1);
      }});
      engineRef.current.shake(capturingKing?18:6,0.35);
      const accent=getComputedStyle(document.documentElement).getPropertyValue('--accent').trim();
      engineRef.current.burst(occupant.sprite.x,occupant.sprite.y,{count:capturingKing?50:20,color:accent,speed:[80,340],size:[2,5],life:[0.4,0.9]});
    }
    piece.row=targetSq.row;
    piece.col=targetSq.col;
    const target=squareToScreen(targetSq.row,targetSq.col,layout);
    /* Trail */
    playMoveTrail(engineRef.current,piece.sprite,fromX,fromY,target.x,target.y,piece.color==='light'?'#8fd46a':'#ff6d43');
    /* Coronación */
    let promoted=false;
    const canCrown=mod.id==='promote'||piece.canCrown;
    if(canCrown&&piece.type==='pawn'){
      const lastRow=piece.color==='light'?0:boardRef.current.rows-1;
      if(piece.row===lastRow){
        promoted=true;
        const style=getComputedStyle(document.documentElement);
        const colorMap={light:'--lightPiece',dark:'--darkPiece',wild:'--wildPiece'};
        const c=style.getPropertyValue(colorMap[piece.color]).trim();
        gsap.to(piece.sprite,{x:target.x,y:target.y,scale:1,duration:opts.byBot?0.4:0.3,ease:'power2.out',onComplete:()=>{
          piece.sprite.z=10;
          playCoronation(engineRef.current,layout,piece.sprite,()=>{
            piece.type='queen';
            SVG_PIECES.load('queen',c).then(img=>{
              if(piece.sprite){
                piece.sprite.image=img;
                gsap.to(piece.sprite,{scale:1,duration:0.3,ease:'elastic.out(1,0.5)'});
              }
            });
          });
        }});
        addLog((opts.byBot?'[BOT] ':'')+'corona');
        setMovesThisTurn(prev=>{
          const next=prev+1;
          if(next>=movesPerTurn){
            setTimeout(()=>{
              const cur=matchRef.current.players[(currentIdxRef.current+1)%matchRef.current.players.length];
              if(cur){
                const threats=getThreatenedOwnPieces(piecesRef.current,cur.color,boardRef.current);
                threatRef.current=threats;setThreatCount(threats.size);
              }
              if(!checkEndGame(piecesRef.current,eliminated))advanceTurn(eliminated);
            },180);
            return prev;
          }
          return next;
        });
        return true;
      }
    }
    gsap.to(piece.sprite,{x:target.x,y:target.y,scale:1,duration:opts.byBot?0.4:0.3,ease:'power2.out',onComplete:()=>{
      piece.sprite.z=10;
      const accent=getComputedStyle(document.documentElement).getPropertyValue('--accent').trim();
      engineRef.current.burst(target.x,target.y,{count:6,color:accent,speed:[30,80],size:[1,3],life:[0.25,0.5]});
    }});
    const tag=opts.byBot?'[BOT] ':'';
    if(capturingKing)addLog(tag+'captura rey');

      /* Efecto dramático al capturar rey */
      playKingCapture(engineRef.current,capturedKingSprite,()=>{});

    if(capturing){
      setEliminated(newElim);
      const remaining=piecesRef.current.filter(p=>p!==occupant);
      setTimeout(()=>{
        const cur=matchRef.current.players[(currentIdxRef.current+1)%matchRef.current.players.length];
        if(cur){
          const threats=getThreatenedOwnPieces(piecesRef.current,cur.color,boardRef.current);
          threatRef.current=threats;setThreatCount(threats.size);
        }
        if(!checkEndGame(remaining,newElim))advanceTurn(newElim);

        const next=prev+1;

              const threats=getThreatenedOwnPieces(piecesRef.current,cur.color,boardRef.current);
              threatRef.current=threats;setThreatCount(threats.size);
            }
            if(!checkEndGame(piecesRef.current,eliminated))advanceTurn(eliminated);
          },180);
          return prev;
        }
        return next;
      });
    }
    return true;

    const canvas=canvasRef.current;
    if(!canvas)return;

    function findPieceAt(px,py){
      const sq=screenToSquare(px,py,layoutRef.current);
      if(!sq)return null;
      return piecesRef.current.find(p=>p.row===sq.row&&p.col===sq.col)||null;

    function clearLegal(){legalRef.current=new Set();legalColorRef.current='legal';}
    function clearInspect(){
      if(inspectRef.current){
        inspectRef.current=null;setInspectInfo(null);
        if(!selectedRef.current)clearLegal();
      }
    }
    function selectPiece(p){
      if(selectedRef.current===p){deselect();return;}
      clearInspect();

        const prev=selectedRef.current;
        gsap.to(prev.sprite,{scale:1,duration:0.12,overwrite:'auto'});
        prev.sprite.z=10;

    function handlePointer(e){
      if(!loaded||paused||endState)return;

          const sq=screenToSquare(x,y,layoutRef.current);
          const moved=sq?executeMoveRef.current(selectedRef.current,sq):false;
          if(!moved)returnHome(selectedRef.current);
          deselect();
          return;

        if(piece.color===cur.color){selectPiece(piece);return;}
        const sq={row:piece.row,col:piece.col};
        const moved=executeMoveRef.current(selectedRef.current,sq);
        if(moved)deselect();
        else {returnHome(selectedRef.current);deselect();inspectPiece(piece);}
        return;
      }
      if(!piece){clearInspect();return;}
      if(piece.color===cur.color){selectPiece(piece);}
      else {
        if(inspectRef.current===piece)clearInspect();
        else inspectPiece(piece);
      }
    }
    function onDown(e){
      if(!loaded||paused||endState)return;
      if(e.pointerType==='mouse'&&e.button!==0)return;
      const cur=matchRef.current.players[currentIdxRef.current];
      if(!cur||cur.kind==='bot')return;
      const {x,y}=getPos(e);
      const piece=findPieceAt(x,y);
      if(piece&&piece.color===cur.color){
        try{canvas.setPointerCapture(e.pointerId);}catch(_){}
        dragRef.current={piece,offsetX:piece.sprite.x-x,offsetY:piece.sprite.y-y,startX:x,startY:y,moved:false,pointerId:e.pointerId};
      }
      handlePointer(e);
    }
    function onMove(e){
      const drag=dragRef.current;
      if(!drag||e.pointerId!==drag.pointerId)return;
      const {x,y}=getPos(e);
      const dist=Math.hypot(x-drag.startX,y-drag.startY);
      if(!drag.moved&&dist>6){
        drag.moved=true;
        if(selectedRef.current&&selectedRef.current!==drag.piece){
          gsap.to(selectedRef.current.sprite,{scale:1,duration:0.12});
          selectedRef.current.sprite.z=10;
          selectedRef.current=null;
        }
        clearInspect();
        gsap.killTweensOf(drag.piece.sprite);
        drag.piece.sprite.scale=1.15;
        drag.piece.sprite.z=100;
        canvas.style.cursor='grabbing';
      }
      if(drag.moved){
        drag.piece.sprite.x=x+drag.offsetX;
        drag.piece.sprite.y=y+drag.offsetY;
      }
    }
    function onUp(e){
      const drag=dragRef.current;
      if(!drag||e.pointerId!==drag.pointerId)return;
      const {x,y}=getPos(e);
      dragRef.current=null;
      canvas.style.cursor='default';
      try{canvas.releasePointerCapture(e.pointerId);}catch(_){}
      if(!drag.moved)return;
      const sq=screenToSquare(x,y,layoutRef.current);
      const moved=sq?executeMoveRef.current(drag.piece,sq):false;
      if(!moved)returnHome(drag.piece);
      else gsap.to(drag.piece.sprite,{scale:1,duration:0.18,overwrite:'auto'});
    }
    function onCancel(e){
      const drag=dragRef.current;
      if(!drag||e.pointerId!==drag.pointerId)return;
      dragRef.current=null;
      canvas.style.cursor='default';
      returnHome(drag.piece);
    }
    function onCtx(e){e.preventDefault();}
    canvas.addEventListener('pointerdown',onDown);
    canvas.addEventListener('pointermove',onMove);
    canvas.addEventListener('pointerup',onUp);
    canvas.addEventListener('pointercancel',onCancel);
    canvas.addEventListener('touchstart',(e)=>e.preventDefault(),{passive:false});
    canvas.addEventListener('contextmenu',onCtx);
    return ()=>{
      canvas.removeEventListener('pointerdown',onDown);
      canvas.removeEventListener('pointermove',onMove);
      canvas.removeEventListener('pointerup',onUp);
      canvas.removeEventListener('pointercancel',onCancel);
      canvas.removeEventListener('contextmenu',onCtx);
    };
  },[loaded,paused,addLog,endState]);
  useEffect(()=>{
    if(!loaded||paused||endState)return;
    const player=players[currentIdx];
    if(!player||player.kind!=='bot')return;
    if(eliminated[player.color])return;
    if(movesThisTurn>=movesPerTurn)return;
    setBotThinking(true);
    const delay=400+Math.random()*350;
    const t=setTimeout(()=>{
      const move=botChooseMove(piecesRef.current,player.color,boardRef.current);
      setBotThinking(false);
      if(move)executeMoveRef.current(move.piece,move.target,{byBot:true});
      else {
        addLog('[BOT] sin movimientos.');
        if(!checkEndGame(piecesRef.current,eliminated))advanceTurn(eliminated);
      }
    },delay);
    return ()=>{clearTimeout(t);setBotThinking(false);};
  },[currentIdx,loaded,paused,players,advanceTurn,addLog,eliminated,endState,movesThisTurn,movesPerTurn]);
  /* Detectar cuando un jugador humano no puede moverse */
  useEffect(()=>{
    if(!loaded||paused||endState)return;
    const player=players[currentIdx];
    if(!player||player.kind!=='human')return;
    if(eliminated[player.color])return;
    const hasMove=hasAnyLegalMove(piecesRef.current,player.color,boardRef.current);
    if(!hasMove){
      const t=setTimeout(()=>{
        addLog(player.name+' no puede moverse.');
        if(!checkEndGame(piecesRef.current,eliminated))advanceTurn(eliminated);
      },400);
      return ()=>clearTimeout(t);
    }
  },[currentIdx,loaded,paused,players,eliminated,endState,advanceTurn,addLog,checkEndGame,movesThisTurn]);
  useEffect(()=>{
    function onKey(e){if(e.key==='Escape'&&!endState)setPaused(p=>!p);}
    window.addEventListener('keydown',onKey);
    return ()=>window.removeEventListener('keydown',onKey);
  },[endState]);
  useEffect(()=>{
    if(window.__fullscreenOnStart===false)return;
    const tryFullscreen=async()=>{
      try{
        const el=document.documentElement;
        if(!document.fullscreenElement&&el.requestFullscreen){await el.requestFullscreen({navigationUI:'hide'});}
      }catch(e){}
    };
    const onClickOnce=()=>{tryFullscreen();document.removeEventListener('click',onClickOnce);};
    document.addEventListener('click',onClickOnce,{once:true});
    tryFullscreen();
    return ()=>document.removeEventListener('click',onClickOnce);
  },[]);
  const handleExit=useCallback(()=>{onExit();},[onExit]);
  const handleAbandon=useCallback(()=>{onAbandon();},[onAbandon]);
  const handleRestart=useCallback(()=>{onRestartNode&&onRestartNode();},[onRestartNode]);
  const handleEndRetry=useCallback(()=>{onRestartNode&&onRestartNode();},[onRestartNode]);
  const handleEndContinue=useCallback(()=>{if(endState&&endState.victory)onWin();},[endState,onWin]);
  const ownedMutList=mutations||[];
  const chaosRemaining=3-(chaosTick%3);
  const showTurnFlash=turnFlashVisible&&!paused&&!endState&&players[currentIdx]&&players[currentIdx].kind==='human';
  return React.createElement('div',{className:'screen active'},
    showTurnFlash&&React.createElement(TurnFlash,{key:turnFlashKey,playerName:players[currentIdx].name,color:players[currentIdx].color}),
    React.createElement('div',{className:'topbar'},
      React.createElement('div',null,
        React.createElement('div',{className:'title'},(isBossNode?'Jefe - ':'Piso '+(node.floor+1)+' - ')+mod.name),
        React.createElement('div',{className:'sub'},players.length+' jugadores')),
      React.createElement('div',{className:'spacer'}),
      React.createElement('button',{className:'icon-btn',onClick:()=>setPaused(true),title:'Pausa'},React.createElement(Icon,{name:'pause',size:16}))
    ),
    React.createElement('div',{className:'game-wrap'},
      React.createElement('canvas',{ref:canvasRef,id:'stage'}),
      React.createElement('aside',{className:'hud'},
        React.createElement('div',null,
          React.createElement('h1',null,'Turno'),
          React.createElement('div',{className:'level-name'},'Piso '+(node.floor+1)+'/'+totalFloors)
        ),
        React.createElement('div',{className:'mod-badge'+(isBossNode?' boss':'')},
          React.createElement('div',{className:'mb-icon'},React.createElement(Icon,{name:isBossNode?'boss':mod.icon,size:16})),
          React.createElement('div',{className:'mb-text'},
            React.createElement('strong',null,mod.name),
            React.createElement('span',null,mod.desc))

      const match={
        playerCount:defaultPlayerCount,

          [{color:'light',kind:'human',name:'Jugador 1'},{color:'dark',kind:'bot',name:'Bot 1'}]:
          [{color:'light',kind:'human',name:'Jugador 1'},{color:'dark',kind:'bot',name:'Bot 1'},{color:'wild',kind:'bot',name:'Bot 2'}]

      goTo('setup');

      return next;
    });

  const handleWin=useCallback(()=>{
    const isFirst=mapState&&Object.keys(mapState.completed).length===0;
    markNodeCompleted(currentNode);
    setCurrentNode(null);setCurrentMatch(null);
    if(isFirst){goTo('map');return;}

    const humanPlayers=(currentMatch&&currentMatch.players||[]).filter(p=>p.kind==='human');
    if(!humanPlayers.length){goTo('map');return;}
    const owned=(mapState&&mapState.mutations)||[];
    const pickers=humanPlayers.map((p,idx)=>({
      color:p.color,
      name:p.name,
      choices:pickMutationChoices(Date.now()+idx*997,owned)
    }));
    setPendingMutations({pickers,index:0,floorLabel:'Piso '+(currentNode.floor+1)+' superado'});
    goTo('mutation');
  },[currentNode,mapState,markNodeCompleted,goTo,currentMatch]);
  const handleMutationPick=useCallback((mutId)=>{
    setPendingMutations(prev=>{
      if(!prev)return null;
      const nextIdx=prev.index+1;
      setMapState(prevState=>{
        if(!prevState)return prevState;
        const mutations=[...(prevState.mutations||[]),mutId];
        const next={...prevState,mutations};
        saveMapState(next);
        return next;
      });
      if(nextIdx>=prev.pickers.length){
        setTimeout(()=>goTo('map'),50);
        return null;
      }
      return {...prev,index:nextIdx};
    });
  },[goTo]);
  const handleRestResolve=useCallback((optId)=>{
    if(optId==='extra_mut'){
      const owned=(mapState&&mapState.mutations)||[];
      const humanPlayers=(currentMatch&&currentMatch.players||[]).filter(p=>p.kind==='human');
      const pickers=(humanPlayers.length?humanPlayers:[{color:'light',name:'Jugador 1'}]).map((p,idx)=>({
        color:p.color,name:p.name,choices:pickMutationChoices(Date.now()+idx*997,owned)
      }));
      markNodeCompleted(currentNode);
      setPendingRest(null);
      setPendingMutations({pickers,index:0,floorLabel:'Piso '+(currentNode.floor+1)+' descanso'});
      goTo('mutation');
      return;
    }
    setMapState(prev=>{
      if(!prev)return prev;
      const bonuses={...(prev.bonuses||{})};
      if(optId==='promote_one')bonuses.promoteOne=true;
      else if(optId==='fury_now')bonuses.furyNow=true;
      const completed={...prev.completed,[currentNode.id]:true};
      const nextFloor=Math.min(prev.map.floors-1,currentNode.floor+1);
      const next={...prev,completed,currentFloor:nextFloor,bonuses};
      saveMapState(next);
      return next;
    });
    setPendingRest(null);
    setCurrentNode(null);setCurrentMatch(null);
    goTo('map');
  },[currentNode,mapState,goTo,markNodeCompleted,currentMatch]);
  const handleBagResolve=useCallback((opt)=>{
    setMapState(prev=>{
      if(!prev)return prev;
      const bonuses={...(prev.bonuses||{})};
      let mutations=prev.mutations||[];
      if(opt.kind==='extra_pawn')bonuses.extraPawn=true;
      else if(opt.kind==='extra_rook')bonuses.extraRook=true;
      else if(opt.kind==='mut_fury')mutations=[...mutations,'fury'];
      else if(opt.kind==='mut'){
        const picks=pickMutationChoices(Date.now(),mutations);
        if(picks.length)mutations=[...mutations,picks[0]];
      }
      const completed={...prev.completed,[currentNode.id]:true};
      const nextFloor=Math.min(prev.map.floors-1,currentNode.floor+1);
      const next={...prev,completed,currentFloor:nextFloor,bonuses,mutations};
      saveMapState(next);
      return next;
    });
    setPendingBag(null);
    setCurrentNode(null);setCurrentMatch(null);
    goTo('map');
  },[currentNode,goTo]);
  const handleRestartNode=useCallback(()=>{setGameKey(k=>k+1);},[]);
  const currentMutations=(mapState&&mapState.mutations)||[];
  const oneShotBonuses=(mapState&&mapState.bonuses)||{};
  const handleConsumeBonuses=useCallback(()=>{
    setMapState(prev=>{
      if(!prev)return prev;
      const b={...(prev.bonuses||{})};
      if(!b.promoteOne&&!b.furyNow&&!b.extraPawn&&!b.extraRook)return prev;
      delete b.promoteOne;delete b.furyNow;delete b.extraPawn;delete b.extraRook;
      const next={...prev,bonuses:b};
      saveMapState(next);
      return next;
    });
  },[]);
  useEffect(()=>{
    if(screen==='game'&&currentMatch)handleConsumeBonuses();
  },[screen,currentMatch,handleConsumeBonuses]);
  const currentPicker=pendingMutations&&pendingMutations.pickers&&pendingMutations.pickers[pendingMutations.index];
  return React.createElement(React.Fragment,null,
    screen==='menu'&&React.createElement(MenuScreen,{onNew:handleNew,onContinue:handleContinue,onSettings:()=>goTo('settings'),onGallery:()=>goTo('gallery'),onHowTo:()=>setShowHowTo(true),hasRun,version:'v1.4'}),
    screen==='gallery'&&React.createElement(GalleryScreen,{onBack:()=>goTo('menu')}),
    screen==='map'&&mapState&&React.createElement(MapScreen,{onBack:()=>goTo('menu'),onSelectNode:handleSelectNode,mapState,onNewRun:handleNewFromMap}),
    screen==='setup'&&currentNode&&React.createElement(SetupScreen,{key:'setup-'+currentNode.id,node:currentNode,defaultPlayerCount,onBack:()=>goTo('map'),onStart:handleStartMatch}),
    screen==='rest'&&currentNode&&pendingRest&&React.createElement(RestScreen,{floorLabel:pendingRest.floorLabel,onResolve:handleRestResolve}),
    screen==='bag'&&currentNode&&pendingBag&&React.createElement(BagScreen,{floorLabel:pendingBag.floorLabel,ownedMutations:currentMutations,onResolve:handleBagResolve}),
    screen==='mutation'&&currentPicker&&React.createElement(MutationPickScreen,{
      key:'mp-'+pendingMutations.index,
      choices:currentPicker.choices,
      ownedMutations:currentMutations,
      floorLabel:pendingMutations.floorLabel,
      activePlayerName:currentPicker.name,
      onPick:handleMutationPick
    }),
    screen==='game'&&currentNode&&currentMatch&&React.createElement(GameScreen,{key:'game-'+currentNode.id+'-'+gameKey,node:currentNode,match:currentMatch,mapState,settings,setSettings,mutations:currentMutations,oneShotBonuses:oneShotBonuses,onExit:handleExitGame,onAbandon:handleAbandonRun,onWin:handleWin,onRestartNode:handleRestartNode}),
    screen==='settings'&&React.createElement(SettingsScreen,{onBack:()=>goTo('menu'),settings,setSettings}),
    showHowTo&&React.createElement(HowToPlayOverlay,{onClose:()=>setShowHowTo(false)})
  );
}
const root=ReactDOM.createRoot(document.getElementById('root'));
root.render(React.createElement(App));
